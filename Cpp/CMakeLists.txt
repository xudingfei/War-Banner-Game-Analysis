cmake_minimum_required(VERSION 3.20)
project(WarBannerMapAlgorithms LANGUAGES CXX)
add_library(mapgen src/mapgen.cpp)
target_include_directories(mapgen PUBLIC include)
target_compile_features(mapgen PUBLIC cxx_std_17)
if(MSVC)
  target_compile_options(mapgen PRIVATE /W4 /EHsc /utf-8)
else()
  target_compile_options(mapgen PRIVATE -Wall -Wextra -Wpedantic)
endif()
add_executable(mapgen_demo src/main.cpp)
target_link_libraries(mapgen_demo PRIVATE mapgen)
add_executable(mapgen_tests tests/tests.cpp)
target_link_libraries(mapgen_tests PRIVATE mapgen)
if(MSVC)
  target_compile_options(mapgen_tests PRIVATE /EHsc)
  target_compile_options(mapgen_demo PRIVATE /EHsc)
endif()
enable_testing()
add_test(NAME MapGenerationAndOpeningProof COMMAND mapgen_tests)
set_tests_properties(MapGenerationAndOpeningProof PROPERTIES TIMEOUT 180)
